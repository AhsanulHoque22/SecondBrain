\section{Heuristic Search — Theoretical Foundations}
%=====================================================================

This section covers the \textbf{theory} that underpins every search-algorithm question in the exam.
The year-by-year traces in later sections apply these concepts to specific graphs.

%----------------------------------------------------------------------
\subsection{BFS (Uninformed) vs.\ Best-First Search (Informed)}
%----------------------------------------------------------------------

\noindent The fundamental divide in search algorithms is between those that use \textbf{no domain knowledge} and those that use \textbf{heuristic guidance}.

\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{p{4cm}|p{6cm}|p{6cm}}
\toprule
\textbf{Criterion} & \textbf{BFS (Uninformed / Blind)} & \textbf{Best-First Search (Informed / Heuristic)} \\
\midrule
\textbf{What guides expansion?} &
  \textbf{Structural position} — depth from the root. BFS expands level by level, regardless of where the goal is. &
  \textbf{Evaluation function} $f(n)$ — incorporates domain knowledge (estimates of cost to the goal). Expands the ``most promising'' node first. \\
\midrule
\textbf{Fringe data structure} &
  \textbf{FIFO queue} — first-in, first-out. Nodes are processed in the order they are discovered. &
  \textbf{Priority queue} — nodes are dequeued in order of $f(n)$ (lowest first). The queue is dynamically reordered as new nodes are discovered. \\
\midrule
\textbf{Knowledge used} &
  \textbf{None} — only the problem definition (initial state, goal test, successor function, step costs). &
  \textbf{Heuristic function} $h(n)$ — estimated cheapest cost from $n$ to the goal. Domain-specific (e.g., straight-line distance, Manhattan distance). \\
\midrule
\textbf{Efficiency} &
  \textbf{Expensive} for large state spaces — $O(b^d)$ time and space. Must explore all nodes at depth $d{-}1$ before checking depth $d$. &
  \textbf{Potentially much faster} — a good heuristic dramatically prunes the search space. But with a poor heuristic, can be as bad as uninformed search. \\
\midrule
\textbf{Completeness} &
  \textbf{Complete} — guaranteed to find a solution if one exists (finite branching factor). &
  \textbf{Depends on the variant} — Greedy BFS is incomplete (can loop). A* is complete. \\
\midrule
\textbf{Optimality} &
  \textbf{Optimal} only for uniform step costs. &
  \textbf{Depends on the variant} — Greedy BFS is not optimal. A* is optimal if $h$ is admissible. \\
\midrule
\textbf{Examples} &
  BFS, UCS (expand by $g(n)$), DFS, IDDFS &
  Greedy Best-First (expand by $h(n)$), A* (expand by $f(n)=g(n)+h(n)$), IDA*, SMA* \\
\bottomrule
\end{tabular}
\end{center}

\begin{notesbox}
\textbf{Key insight:} ``Uninformed'' does \emph{not} mean ``stupid'' — it means the algorithm works for \textbf{any} problem without modification. ``Informed'' means the algorithm exploits \textbf{domain-specific} knowledge to work more efficiently, but the heuristic must be designed for each problem. An uninformed algorithm needs no setup; an informed algorithm needs a clever heuristic.
\end{notesbox}

%----------------------------------------------------------------------
\subsection{Informed vs.\ Uninformed Search — The Fundamental Divide}
%----------------------------------------------------------------------

\noindent Every search algorithm belongs to one of two families, distinguished by a single question: \textbf{Does it use domain-specific knowledge beyond the problem definition?}

\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{p{4.5cm}|p{5.5cm}|p{5.5cm}}
\toprule
\textbf{Criterion} & \textbf{Uninformed (Blind) Search} & \textbf{Informed (Heuristic) Search} \\
\midrule
\textbf{Knowledge used} &
  Only the problem definition: initial state, goal test, successor function, step costs. \textbf{No estimate} of how close a state is to the goal. &
  Uses a \textbf{heuristic function} $h(n)$ — domain-specific knowledge estimating the cheapest cost from $n$ to the goal. \\
\midrule
\textbf{How nodes are chosen} &
  By \textbf{structural criteria} — depth (BFS, DFS, IDDFS), path cost so far (UCS), or symmetry (Bidirectional). &
  By an \textbf{evaluation function} $f(n)$ combining $g(n)$ and $h(n)$ — the algorithm steers toward promising-looking nodes. \\
\midrule
\textbf{Efficiency} &
  \textbf{Exponential} in the worst case. Can explore vast portions of the state space with no guidance. &
  \textbf{Potentially far more efficient} — a good heuristic prunes large unpromising regions. With a perfect heuristic, A* expands only nodes on the optimal path. \\
\midrule
\textbf{Generality} &
  \textbf{Universal} — works for \emph{any} problem without modification. No domain knowledge needed. Zero setup cost. &
  \textbf{Problem-specific} — the heuristic must be designed (or learned) for each problem. \\
\midrule
\textbf{Optimality guarantee} &
  UCS guarantees optimality for any non-negative costs; BFS/IDDFS only for uniform costs. &
  A* guarantees optimality if $h$ is admissible ($h \leq h^*$). Greedy BFS gives no optimality guarantee. \\
\midrule
\textbf{When to use} &
  Problem is new/unknown; no good heuristic exists; state space is small; or as a fallback. &
  Domain knowledge is available; an admissible heuristic can be defined; state space is large; optimality matters. \\
\midrule
\textbf{Examples} &
  BFS, DFS, DLS, IDDFS, UCS, Bidirectional Search &
  Greedy BFS, A*, IDA*, RBFS, SMA*, Beam Search \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.4em}

\noindent\textbf{The continuum — not a hard binary:} Some algorithms blur the line.
\begin{itemize}[leftmargin=1.5cm, itemsep=2pt, topsep=3pt]
  \item \textbf{UCS} uses $g(n)$ — this is ``cost-aware'' but not ``goal-aware.'' It knows how much you've spent but has no idea how much remains. It is technically uninformed because $g(n)$ is derived from the problem definition (step costs), not from domain knowledge.
  \item \textbf{Bidirectional Search} uses two simultaneous searches (forward from start, backward from goal). It is structurally informed — leveraging the fact that two smaller searches are cheaper than one large one — but uses no heuristic. It is formally uninformed.
  \item \textbf{A*} sits at the intersection: $f(n) = g(n) + h(n)$. The $g(n)$ part is uninformed (known costs); the $h(n)$ part is informed (domain estimate). This is why A* subsumes UCS (when $h(n){=}0$) and Greedy (conceptually, when $g(n)$ is ignored).
\end{itemize}

\subsection{Progression of Uninformed Search Algorithms}
%----------------------------------------------------------------------

\begin{solutionbox}
\textbf{Question:} Discuss how uninformed search algorithms can be seen as an evolution of ideas, where each new algorithm is designed to address a limitation of an earlier one. Include the progression among: BFS, DFS, DLS, IDDFS, UCS, and Bidirectional Search.
\end{solutionbox}

\vspace{0.5em}

\noindent Uninformed search algorithms did not emerge in isolation — they form a \textbf{logical progression}, each motivated by a specific weakness of its predecessor. Understanding this progression turns six seemingly disconnected algorithms into one coherent story.

\subsubsection*{Act I — The Foundational Trade-off: BFS vs.\ DFS}

\noindent The two simplest algorithms represent opposite ends of a fundamental trade-off.

\begin{center}
\begin{tikzpicture}[
  box/.style={draw, rounded corners=4pt, minimum width=5cm, minimum height=2.8cm, align=left, fill=blue!4, font=\small},
  arrow/.style={-{Stealth[length=6pt, width=5pt]}, thick}
]
\node[box, fill=green!6] (bfs) at (-4, 0) {
  \textbf{BFS (Breadth-First)}\\[2pt]
  \checkmark\ Complete — finds solution if one exists.\\
  \checkmark\ Optimal for uniform costs.\\
  \texttimes\ \textbf{$O(b^d)$ space} — must store entire frontier.\\
  \hspace{4pt} For $b{=}10, d{=}10$: 10 billion nodes in memory.
};

\node[box, fill=red!6] (dfs) at (4, 0) {
  \textbf{DFS (Depth-First)}\\[2pt]
  \texttimes\ \textbf{Not complete} — infinite branches trap it.\\
  \texttimes\ Not optimal — finds deep solutions first.\\
  \checkmark\ \textbf{$O(bm)$ space} — stores only current path.\\
  \hspace{4pt} For $b{=}10, d{=}10$: $\sim$100 nodes in memory.
};

\draw[arrow, bend left=20, color=orange!70!black] (bfs) to node[midway, above, font=\footnotesize\bfseries, text=orange!70!black] {swap for space} (dfs);
\draw[arrow, bend left=20, color=orange!70!black] (dfs) to node[midway, below, font=\footnotesize\bfseries, text=orange!70!black] {swap for completeness} (bfs);
\end{tikzpicture}
\end{center}

\noindent\textbf{The dilemma:} BFS guarantees the answer but needs impossible amounts of memory. DFS uses negligible memory but may never find the answer. The entire subsequent evolution is about \textbf{getting the best of both}.

\subsubsection*{Act II — DFS's First Fix: Depth-Limited Search (DLS)}

\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{p{3.5cm}|p{6cm}|p{6cm}}
\toprule
\textbf{Motivation} & \textbf{What DLS does} & \textbf{Remaining problem} \\
\midrule
DFS can descend \textbf{infinitely deep} on an infinite branch, never backtracking to find a shallow solution. &
DLS adds a \textbf{depth cutoff} $L$: DFS stops at depth $L$ and backtracks. This \textbf{prevents infinite descent} — the search is now bounded. &
\textbf{How to choose $L$?} If $L < d$ (solution depth), DLS returns \textbf{failure} even though a solution exists. If $L \gg d$, DLS wastes time. DLS is \textbf{not complete} unless you already know $d$. \\
\bottomrule
\end{tabular}
\end{center}

\noindent\textbf{The contribution:} DLS showed that adding a depth bound fixes DFS's completeness problem — but only if you know the right bound. The question becomes: \emph{how to find $d$ without knowing it in advance?}

\subsubsection*{Act III — Finding the Right Depth: Iterative Deepening DFS (IDDFS)}

\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{p{3.5cm}|p{6cm}|p{6cm}}
\toprule
\textbf{Motivation} & \textbf{What IDDFS does} & \textbf{Remaining problem} \\
\midrule
DLS fails if $L < d$ (misses solution) or wastes time if $L \gg d$. You don't know $d$ in advance. &
IDDFS runs DLS \textbf{repeatedly} with $L = 0, 1, 2, \ldots$ until the goal is found. This guarantees finding the shallowest solution — \textbf{completeness restored}. &
\textbf{Non-uniform costs ignored.} IDDFS finds the shallowest goal (fewest steps), not the cheapest. If edge costs vary, the shortest path in steps may be expensive. Nodes near the root are re-expanded — overhead factor $\frac{b}{b-1}$. \\
\bottomrule
\end{tabular}
\end{center}

\noindent\textbf{The contribution:} IDDFS solved the BFS-vs-DFS dilemma for uniform-cost problems: $O(bd)$ space (from DFS) $+$ completeness $+$ shallowest-solution optimality (from BFS). The repeated-work overhead is only $\sim$11\% for $b{=}10$ — an acceptable price. \textbf{This is the best uninformed algorithm for uniform-cost problems with large state spaces.}

\subsubsection*{Act IV — Handling Non-Uniform Costs: Uniform-Cost Search (UCS)}

\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{p{3.5cm}|p{6cm}|p{6cm}}
\toprule
\textbf{Motivation} & \textbf{What UCS does} & \textbf{Remaining problem} \\
\midrule
BFS and IDDFS assume \textbf{all steps cost the same}. In the real world, actions have different costs (mountain roads cost more than highways). Neither shallowest-path algorithm can find the cheapest path with non-uniform costs. &
UCS expands nodes by \textbf{lowest path cost $g(n)$}, not depth. It is \textbf{optimal for any non-negative edge costs} — the first time it expands the goal, the cheapest path is found. &
\textbf{No guidance toward the goal.} UCS expands equally in \emph{all} directions — it knows what you've spent but has \textbf{no idea how much remains}. In a large graph, UCS may explore enormous regions away from the goal. \\
\bottomrule
\end{tabular}
\end{center}

\noindent\textbf{The contribution:} UCS generalised optimality from ``shallowest'' to ``cheapest.'' But it exposed a new problem: without knowledge of where the goal is, the algorithm is undirected — it wastes effort exploring in wrong directions. This limitation \textbf{directly motivates informed search} — add a heuristic $h(n)$ to estimate remaining distance, and UCS becomes A*.

\subsubsection*{Act V — Exploiting Symmetry: Bidirectional Search}

\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{p{3.5cm}|p{6cm}|p{6cm}}
\toprule
\textbf{Motivation} & \textbf{What Bidirectional does} & \textbf{Remaining problem} \\
\midrule
A single-direction search explores $O(b^d)$ nodes. But a search backward from the goal is equally valid — and two searches meeting in the middle each explore only $O(b^{d/2})$ nodes. This is an \textbf{exponential reduction}: $b^d \to 2 \cdot b^{d/2}$. &
Run two simultaneous searches — one forward from start, one backward from goal(s). The solution is found when the two frontiers intersect. Can use BFS, UCS, or even A* as the underlying search method. &
\textbf{Requires a reversible successor function.} The backward search needs a predecessor function — ``what states lead \emph{to} the goal?'' This is not always easy to define. Also, checking frontier intersection at every step adds overhead. For some problems (theorem proving), the backward branching factor is much larger. \\
\bottomrule
\end{tabular}
\end{center}

\noindent\textbf{The contribution:} Bidirectional search exploits problem \textbf{symmetry} — the fact that searching from the goal backward is just as valid as from the start forward. It is still uninformed, but achieves an exponential speedup through structural cleverness alone. Combined with IDDFS (bidirectional IDDFS) or A* (bidirectional A*), it becomes even more powerful.

\subsubsection*{The Evolutionary Narrative (Exam Summary)}

\begin{enumerate}[leftmargin=1.5cm, itemsep=5pt, topsep=3pt]
  \item \textbf{BFS} — Gives us completeness and optimality, but at the impossible price of $O(b^d)$ memory. \textbf{Starting point.}

  \item \textbf{DFS} — Flips the trade-off: $O(bm)$ memory is practical, but the algorithm may never find a solution. \textbf{Motivated by BFS's memory explosion.}

  \item \textbf{DLS} — Fixes DFS's infinite-descent problem with a depth cutoff. But now you must guess the cutoff — guess too low and it reports failure when a solution exists. \textbf{Motivated by DFS's incompleteness.}

  \item \textbf{IDDFS} — Eliminates the guesswork: $L=0,1,2,\ldots$ until success. Restores completeness $+$ shallowest-solution optimality while keeping $O(bd)$ space — the best of BFS and DFS. \textbf{Motivated by DLS's dependence on unknown $L$.}

  \item \textbf{UCS} — Abandons depth for path cost $g(n)$. Finds the cheapest path regardless of non-uniform edge costs. But expands blindly in all directions. \textbf{Motivated by IDDFS/BFS's uniform-cost assumption.}

  \item \textbf{Bidirectional Search} — Attacks exponential complexity directly: two $d/2$-depth searches replace one $d$-depth search. Orthogonal improvement — can be layered on top of BFS, UCS, or A*. \textbf{Motivated by single-direction search's $O(b^d)$ cost.}
\end{enumerate}

\begin{notesbox}
\textbf{The arc to informed search:} The progression above solves structural problems (memory, completeness, cost-sensitivity) but never addresses the \textbf{core inefficiency} — none of these algorithms knows where the goal is. The next evolutionary step — \textbf{informed search} (Greedy, A*, IDA*) — adds a heuristic $h(n)$ that \emph{estimates} remaining distance. UCS with $h(n)$ becomes A*; IDDFS with $h(n)$ becomes IDA*. The progression continues, now with domain knowledge.
\end{notesbox}

\subsection{Heuristic Search — Introduction \& the Best-First Family}
%----------------------------------------------------------------------

\subsubsection*{What Makes a Search ``Informed''?}

\noindent An \textbf{informed} (or heuristic) search algorithm uses \textbf{problem-specific knowledge} beyond the problem definition to guide the search toward the goal more efficiently. This knowledge is encoded in a \textbf{heuristic function} $h(n)$.

\vspace{0.4em}
\noindent\textbf{Generic Best-First Search Algorithm:}

\begin{enumerate}[leftmargin=1.5cm, itemsep=4pt]
  \item Initialise the fringe (priority queue) with the start node.
  \item \textbf{Loop:} Remove the node with the \textbf{lowest evaluation function} $f(n)$ from the fringe.
  \item If that node is the goal, return the solution path.
  \item Expand the node — generate all successors.
  \item For each successor: compute $f(\text{successor})$, insert into the fringe (priority queue).
  \item Repeat until the fringe is empty (failure).
\end{enumerate}

\noindent\textbf{The Best-First Search Family:} The ``family'' is unified by the priority-queue fringe — the difference is \textbf{what} $f(n)$ measures:

\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{l|l|l|l}
\toprule
\textbf{Algorithm} & \textbf{Evaluation $f(n)$} & \textbf{Type} & \textbf{Key property} \\
\midrule
\textbf{UCS} &
  $f(n) = g(n)$ &
  Uninformed &
  Optimal for any non-negative costs. \\
\textbf{Greedy BFS} &
  $f(n) = h(n)$ &
  Informed &
  Fast but not optimal. Uses only heuristic. \\
\textbf{A*} &
  $f(n) = g(n) + h(n)$ &
  Informed &
  Optimal if $h$ is admissible. Best-known informed algorithm. \\
\textbf{IDA*} &
  $f(n) = g(n) + h(n)$ &
  Informed &
  Like A* but $O(bd)$ space. Uses iterative deepening with $f$-cutoff. \\
\textbf{RBFS} &
  $f(n) = g(n) + h(n)$ &
  Informed &
  Recursive Best-First — linear space, slightly more overhead than IDA*. \\
\textbf{SMA*} &
  $f(n) = g(n) + h(n)$ &
  Informed &
  Simplified Memory-Bounded A* — uses all available memory, drops worst nodes when full. \\
\bottomrule
\end{tabular}
\end{center}

\begin{notesbox}
\textbf{Exam relevance:} The CUET exam tests UCS, Greedy BFS, A*, and IDA*/IDDFS — these are the four you must be able to trace step-by-step on a given graph. RBFS and SMA* are mentioned in Russell \& Norvig but have never appeared in a CUET past paper (2020--2024).
\end{notesbox}

%----------------------------------------------------------------------
\subsection{The Heuristic Function $h(n)$}
%----------------------------------------------------------------------

\subsubsection*{Definition}

\[
\boxed{h(n) = \text{estimated cheapest cost from node } n \text{ to a goal node}}
\]

\vspace{0.4em}
\noindent\textbf{Key properties:}
\begin{itemize}[leftmargin=1.5cm, itemsep=2pt, topsep=3pt]
  \item $h(n)$ is \textbf{problem-specific} — you design it using domain knowledge. The same generic search code works for any problem, but $h(n)$ encodes what you know about \emph{this} problem.
  \item $h(n) \geq 0$ for all $n$, and $h(G) = 0$ for any goal node $G$.
  \item $h(n)$ is \textbf{not} the action cost — it is an \textbf{estimate} of the remaining cost.
  \item A \textbf{perfect heuristic} satisfies $h(n) = h^*(n)$ for all $n$ (where $h^*(n)$ is the true optimal cost). With a perfect heuristic, A* expands only nodes on optimal paths — it does zero wasted work.
\end{itemize}

\subsubsection*{Example — Romania Map with Straight-Line Distance Heuristic}

\noindent The classic illustration from Russell \& Norvig (Chapter~3). The problem: drive from \textbf{Arad} to \textbf{Bucharest} via the Romanian road network. The heuristic $h_{\text{SLD}}(n)$ is the \textbf{straight-line (as-the-crow-flies) distance} from city $n$ to Bucharest — the shortest possible distance, ignoring roads.

\vspace{0.6em}

\begin{center}
\begin{tikzpicture}[
  every node/.style={draw, circle, minimum size=0.65cm, inner sep=0pt, font=\footnotesize\bfseries},

\subsection{Greedy Best-First Search — Algorithm, Example, Properties}
%----------------------------------------------------------------------

\subsubsection*{Algorithm}

\noindent Greedy Best-First Search uses \textbf{only the heuristic} as its evaluation function:

\[
\boxed{f(n) = h(n)}
\]

\vspace{0.3em}
\noindent At every step, expand the fringe node with the \textbf{lowest heuristic value} $h(n)$. The fringe is a \textbf{min-priority queue sorted by $h(n)$}. Path cost $g(n)$ is \textbf{completely ignored}.

\subsubsection*{Romania Trace — How Greedy Finds a Suboptimal Path}

\noindent Problem: drive from \textbf{Arad} to \textbf{Bucharest} using $h_{\text{SLD}}$.

\noindent\textbf{Greedy path:} Arad $\xrightarrow{140}$ Sibiu $\xrightarrow{99}$ Fagaras $\xrightarrow{211}$ Bucharest
$\quad\Rightarrow\quad \textbf{Cost = 450}$

\noindent\textbf{Optimal path (A* finds this; Greedy misses it):} Arad $\xrightarrow{140}$ Sibiu $\xrightarrow{80}$ Rimnicu $\xrightarrow{97}$ Pitesti $\xrightarrow{101}$ Bucharest
$\quad\Rightarrow\quad \textbf{Cost = 418}$

\begin{answerbox}
\textbf{Why Greedy failed — the root cause:}

At Sibiu, Greedy compared two routes eastward:
\begin{itemize}[noitemsep]
  \item \textbf{Fagaras:} $h{=}176$ — looks just 176 km from Bucharest as the crow flies!
  \item \textbf{Rimnicu:} $h{=}193$ — looks slightly further.
\end{itemize}

Greedy chose Fagaras because $176 < 193$, \textbf{completely ignoring} that:
\begin{itemize}[noitemsep]
  \item The road from Fagaras to Bucharest is 211 km (mountains force a winding route — $h$ underestimates by 35 km).
  \item The Rimnicu $\to$ Pitesti $\to$ Bucharest route costs $80+101=181$ km total — $h$ for Rimnicu is only off by 0 (it's perfect!) and Pitesti is off by just 1.
\end{itemize}

\textbf{Moral:} A low $h(n)$ can be \textbf{deceptive}. A node that looks close to the goal may have a very expensive road ahead. Greedy's refusal to track how much it has already spent ($g(n)$) is its fatal flaw.
\end{answerbox}

\subsubsection*{Properties of Greedy Best-First Search}

\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{l|p{7.5cm}|p{7.5cm}}
\toprule
\textbf{Property} & \textbf{Status} & \textbf{Explanation} \\
\midrule
\textbf{Completeness} &
  \texttimes\ \textbf{Not complete} &
  Can get stuck in infinite loops — if the heuristic guides it into a cycle, it may never escape. A closed set fixes this but costs $O(b^d)$ memory. On finite graphs with a closed set, it is complete. \\
\midrule
\textbf{Optimality} &
  \texttimes\ \textbf{Not optimal} &
  Ignores $g(n)$ — will take a cheap-looking path even if the total cost is high. Romania: cost 450 vs.\ optimal 418. \\
\midrule
\textbf{Time} &
  $O(b^m)$ worst case &
  Can explore the entire tree with a poor heuristic. With a good heuristic, extremely fast — Romania: only 4 expansions. \\
\midrule
\textbf{Space} &
  $O(b^m)$ &
  Stores all generated nodes in the priority queue. Good heuristic $\Rightarrow$ fewer nodes generated. \\
\midrule
\textbf{Sensitivity} &
  \textbf{Very high} — entirely dependent on $h$ quality. &
  With perfect $h=h^*$: optimal in $O(d)$ expansions. With $h=0$ everywhere: degenerates to random selection from the fringe. \\
\bottomrule
\end{tabular}
\end{center}

\subsection{Greedy vs.\ A* — Direct Comparison}
%----------------------------------------------------------------------

\noindent Both are \textbf{informed search algorithms} using a priority queue. The critical difference is \textbf{what they optimise}:

\[
\text{Greedy: } f(n) = h(n) \qquad\qquad \text{A*: } f(n) = g(n) + h(n)
\]

\begin{center}
\renewcommand{\arraystretch}{1.6}
\begin{tabular}{l|p{6.5cm}|p{6.5cm}}
\toprule
\textbf{Criterion} & \textbf{Greedy Best-First} & \textbf{A* Search} \\
\midrule
\textbf{Priority} &
  Lowest $h(n)$ only — ``Which node \emph{looks} closest to the goal?'' &
  Lowest $g(n)+h(n)$ — ``Which node gives the cheapest estimated total path?'' \\
\midrule
\textbf{Completeness} &
  \texttimes\ Not complete (can loop without closed set) &
  \checkmark\ \textbf{Complete} on finite graphs \\
\midrule
\textbf{Optimality} &
  \texttimes\ \textbf{Not optimal} — ignores past cost &
  \checkmark\ \textbf{Optimal} if $h$ is admissible ($h \leq h^*$) \\
\midrule
\textbf{Time} &
  $O(b^m)$ worst case; very fast with good $h$ &
  $O(b^d)$ worst case; $O(d)$ with perfect $h$ \\
\midrule
\textbf{Space} &
  $O(b^m)$ — stores fringe &
  $O(b^d)$ — stores fringe (A*'s main practical weakness) \\
\midrule
\textbf{Heuristic req.} &
  None (works with any $h$) &
  \textbf{Must be admissible} for optimality; consistent for efficiency on graphs \\
\midrule
\textbf{Romania result} &
  4 expansions, cost 450 (suboptimal) &
  6 expansions, cost 418 (optimal) \\
\midrule
\textbf{Use when} &
  Speed $>$ optimality; excellent heuristic; large state space &
  Optimality required; admissible heuristic; state space fits in memory \\
\bottomrule
\end{tabular}
\end{center}

\subsubsection*{Algorithm 1 — Uniform-Cost Search (UCS)}
%----------------------------------------------------------------

\textbf{Key idea:} Always expand the node with the \textbf{lowest path cost} $g(n)$ from the start.
The fringe is a \textbf{min-priority queue sorted by} $g(n)$.
A node is closed (never re-expanded) once popped.
If a \emph{cheaper} path to a fringe node is found, its cost is updated (replaced).

\textbf{Fringe notation:} Node$[g]$ sorted ascending by $g$.

\end{answerbox}

%----------------------------------------------------------------
\subsubsection*{Algorithm 3 — A* Search}
%----------------------------------------------------------------

\textbf{Key idea:} Expand the node with the \textbf{lowest evaluation function}
$f(n) = g(n) + h(n)$, where $g(n)$ = exact cost from start and $h(n)$ = heuristic estimate to goal.
The fringe is sorted by $f(n)$ (ascending). Ties in $f$ broken alphabetically.
If a cheaper path to a fringe node is found, update its entry.

\textbf{Fringe notation:} Node$[g, h, f]$ sorted ascending by $f$.

\subsubsection*{Algorithm 4 — Iterative Deepening DFS (IDDFS)}

Nodes on the \textbf{current path} (ancestors) are not revisited to avoid infinite loops.

\subsubsection*{(i) Robot Vacuum Cleaning Problem (Indoor Environment)}

\begin{description}[leftmargin=3.2cm, labelwidth=3cm, labelsep=0.2cm]
  \item[\textbf{State:}]
    A pair $(\,l,\; D\,)$ where $l$ is the robot's current room and $D \subseteq \{\text{all rooms}\}$ is the set of dirty rooms.
    \emph{Example:} $(\text{Kitchen},\; \{\text{Bedroom, Lounge}\})$.

  \item[\textbf{Initial state:}]
    $(\text{start room},\; \{\text{all currently dirty rooms}\})$.

  \item[\textbf{Goal test:}]
    $D = \emptyset$ \quad (no dirty rooms remain).

  \item[\textbf{Operators:}]
    \begin{enumerate}[noitemsep]
      \item $\mathrm{Move}(r_i \to r_j)$: Move from room $r_i$ to adjacent room $r_j$ (changes location; $D$ unchanged).
      \item $\mathrm{Suck}()$: Vacuum the current room (removes current room from $D$ if dirty).
    \end{enumerate}
    Static obstacles (furniture) constrain which rooms are adjacent.

  \item[\textbf{Path cost:}]
    Total energy used: each $\mathrm{Move}$ costs $c_{\mathrm{move}}$ energy units; each $\mathrm{Suck}$ costs $c_{\mathrm{suck}}$ units. Goal: minimise total energy.

  \item[\textbf{Informed vs.\ Uninformed:}]
    \textbf{Informed search (A*) is more suitable.} A strong heuristic $h(n) = |D| \times \overline{d}$ (number of remaining dirty rooms multiplied by average travel cost between rooms) estimates remaining work without overestimating. With many rooms, uninformed BFS/UCS must explore exponentially many $(l, D)$ state combinations blindly. A* with a good heuristic dramatically prunes this search space.
\end{description}

\begin{description}[leftmargin=3.2cm, labelwidth=3cm, labelsep=0.2cm]
  \item[\textbf{State:}]

  \item[\textbf{Initial state:}]
    $(0,\; 0)$ — both jugs empty.

  \item[\textbf{Goal test:}]
    $x = 4$ (exactly 4 litres in the 5L jug). The 3L jug cannot hold 4L, so only $x=4$ satisfies the goal.

  \item[\textbf{Operators:}]
    \begin{enumerate}[noitemsep]
      \item \textbf{Fill 5L:} $(x,y) \to (5,\;y)$
      \item \textbf{Fill 3L:} $(x,y) \to (x,\;3)$
      \item \textbf{Empty 5L:} $(x,y) \to (0,\;y)$
      \item \textbf{Empty 3L:} $(x,y) \to (x,\;0)$
      \item \textbf{Pour 5L $\to$ 3L:} $(x,y) \to (x{-}d,\;y{+}d)$ where $d = \min(x,\; 3{-}y)$
      \item \textbf{Pour 3L $\to$ 5L:} $(x,y) \to (x{+}d,\;y{-}d)$ where $d = \min(y,\; 5{-}x)$

  \item[\textbf{State space:}]

    (0,0)

    \xrightarrow{\mathrm{Pour\,5\to3}} (2,3)
    \xrightarrow{\mathrm{Empty\,3L}} (2,0)

    \xrightarrow{\mathrm{Fill\,5L}} (5,2)
    \xrightarrow{\mathrm{Pour\,5\to3}} (4,3)
    \quad \Rightarrow x=4 \;\checkmark
    \]
\end{description}

\subsubsection*{(i) Travelling Salesman Problem (TSP)}

\begin{description}[leftmargin=3.6cm, labelwidth=3.4cm, labelsep=0.2cm]
  \item[\textbf{State:}]
    A pair $(\,\ell,\; V\,)$ where $\ell$ is the current city and $V \subseteq \{\text{all } N \text{ cities}\}$ is the set of \emph{unvisited} cities.
    \emph{Example midway:} $(\text{Dhaka},\; \{\text{Chittagong, Sylhet}\})$ means we are in Dhaka and still need to visit Chittagong and Sylhet.

  \item[\textbf{Initial state:}]

%----------------------------------------------------------------
\subsubsection*{Algorithm 1 — Uniform-Cost Search (UCS)}

\textbf{Key idea:} Expand the node with the \textbf{lowest path cost} $g(n)$. Fringe is a min-priority queue sorted by $g(n)$.

\textbf{Closed set:} $\{\text{nodes already expanded}\}$.

\end{tabular}

\begin{answerbox}
\textbf{UCS Solution:} $A \;\xrightarrow{1}\; C \;\xrightarrow{2}\; D \;\xrightarrow{4}\; E \;\xrightarrow{2}\; G$
\hfill \textbf{Total cost: $1+2+4+2 = \mathbf{9}$}

Nodes expanded (in order): A, C, D, B, E, F, G \quad(7 nodes total)

UCS guarantees the optimal path because it explores in order of increasing $g(n)$ and step costs are non-negative.
\end{answerbox}

\textbf{Key idea:} Expand the node with the \textbf{lowest heuristic} $h(n)$. Fringe sorted by $h$ ascending.
Path cost $g(n)$ is \emph{completely ignored}. This is fast but not guaranteed optimal.

  & C[\textbf{6}],\; B[\textbf{8}] \\

2 & \textbf{C} & 6

Nodes expanded: A, C, F, G \quad(only 4 nodes — extremely fast)

Ties in $f$ broken alphabetically. Closed set prevents re-expansion.

\hfill \textbf{Total cost: $1+2+4+2 = \mathbf{9}$} (optimal!)

Nodes expanded: A, C, D, E, G \quad(only 5 nodes — fewer than UCS's 7)

\textbf{Why does A* outperform UCS here?} The heuristic $h(n)$ provides a useful lower bound. After expanding A and C, A* sees that D has $f{=}8$ (promising — $h(D){=}5$ is close to the true remaining cost of 6) while B has $f{=}12$ (unpromising — $h(B){=}8$ overestimates its potential). A* expands D next and finds the optimal path without ever touching B. UCS, lacking a heuristic, expands B unnecessarily at step 4 before reaching the goal.

\textbf{Admissibility check:} Is $h(n)$ admissible? Let's verify:
\begin{itemize}[noitemsep,topsep=3pt]
  \item $h(A){=}8 \leq h^*(A){=}9$ \checkmark
  \item $h(C){=}6 \leq h^*(C){=}8$ \checkmark
  \item $h(D){=}5 \leq h^*(D){=}6$ \checkmark
  \item $h(E){=}1 \leq h^*(E){=}2$ \checkmark
  \item $h(B){=}8 \leq h^*(B){=}5$? \textbf{No!} $h^*(B) = \min(3{+}4{+}2,\; 8{+}2) = 9$. Wait — $B{\to}D{\to}E{\to}G = 3{+}4{+}2=9$, so $h(B){=}8 < 9$ \checkmark. And $B{\to}E{\to}G = 8{+}2=10$. So $h^*(B){=}9$, $h(B){=}8$ is admissible.
  \item $h(F){=}4 \leq h^*(F){=}8$ \checkmark
\end{itemize}
\textbf{Yes, the heuristic is admissible} — $h(n) \leq h^*(n)$ for all nodes. This is why A* found the optimal solution.
\end{notesbox}

\subsection{Part c — IDDFS: Synthesising Benefits of BFS and DFS \ [2 marks]}
%----------------------------------------------------------------------

\begin{solutionbox}
\textbf{Question:} Illustrate with an appropriate example how IDDFS synthesises the benefits of both BFS and DFS — achieving optimality, completeness, and linear space complexity. Nevertheless, time complexity deteriorates in comparison to both BFS and DFS.
\end{solutionbox}

\vspace{0.6em}

\subsubsection*{The Core Idea}

IDDFS (Iterative Deepening Depth-First Search) runs DFS repeatedly with an increasing depth cutoff: first $L{=}0$, then $L{=}1$, $L{=}2$, \ldots, until the goal is found. At each iteration, the search is a standard depth-limited DFS from the root.

\subsubsection*{Illustrative Example}

Consider the following binary tree where G (shaded) is the shallowest goal at depth 3:

\begin{center}
\begin{tikzpicture}[
  level 1/.style={sibling distance=4.2cm, level distance=1.3cm},
  level 2/.style={sibling distance=2.1cm, level distance=1.3cm},
  level 3/.style={sibling distance=1.0cm, level distance=1.2cm},
  every node/.style={draw, circle, minimum size=0.6cm, inner sep=0pt, font=\footnotesize\bfseries},
  goalnode/.style={fill=goalgreen},
  edgelbl/.style={draw=none, font=\footnotesize, inner sep=1pt}
]
\node {A}
  child { node {B}
    child { node {D}
      child { node[goalnode] {G} }
      child { node {H} }
    }
    child { node {E} }
  }
  child { node {C}
    child { node {F} }
    child { node {I} }
  };
\end{tikzpicture}
\end{center}

\vspace{0.4em}

\noindent\textbf{IDDFS Execution:}

\begin{center}
\renewcommand{\arraystretch}{1.2}
\small
\begin{tabular}{c|l|p{9cm}}
\toprule
\textbf{L} & \textbf{Nodes visited (DFS order)} & \textbf{Outcome} \\
\midrule
0 & A & A $\neq$ G. No solution at depth 0. \\
1 & A, B, C & Neither B nor C is G. No solution at depth 1. \\
2 & A, B, D, E, C, F, I & No goal at depth 2. \\
3 & A, B, D, \textbf{G} $\leftarrow$ \goal & \textbf{Found!} Shallowest goal at depth 3. \\
\bottomrule
\end{tabular}
\end{center}

\subsubsection*{How IDDFS Inherits Each Algorithm's Strengths}

\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{l|p{5cm}|p{5cm}|p{5cm}}
\toprule
\textbf{Property} & \textbf{BFS} & \textbf{DFS} & \textbf{IDDFS} \\
\midrule
\textbf{Completeness} &
  \checkmark\ Complete (finds solution if one exists) &
  \texttimes\ Incomplete (may loop forever on infinite branches) &
  \checkmark\ \textbf{Complete} — like BFS. Depth-limited DFS avoids infinite loops because it backtracks at the cutoff. \\
\midrule
\textbf{Optimality} &
  \checkmark\ Optimal for uniform step costs (finds shallowest solution) &
  \texttimes\ Not optimal (may find a deep solution first) &
  \checkmark\ \textbf{Optimal for uniform costs} — like BFS. Because it explores all nodes at depth $d{-}1$ before any at depth $d$, the first solution found is the shallowest. \\
\midrule
\textbf{Space} &
  \texttimes\ $O(b^d)$ — stores entire frontier. For $b{=}10,d{=}10$, that's 10 billion nodes. &
  \checkmark\ $O(bm)$ — stores only one path + siblings. &
  \checkmark\ \textbf{$O(bd)$} — like DFS. Only the current path (depth $d$) plus $b{-}1$ siblings per level are in memory. For $b{=}10,d{=}10$: $\approx$ 100 nodes. \\
\midrule
\textbf{Time} &
  $O(b^d)$ — visits each node once &
  $O(b^m)$ — may go very deep, but each node visited once &
  \texttimes\ $O(b^d)$ — \textbf{worse constant factor.} Nodes near the root are revisited $d$ times. Overhead factor: $\frac{b}{b-1}$ (e.g.\ for $b{=}10$, about $1.11\times$ BFS). \\
\bottomrule
\end{tabular}
\end{center}

\subsubsection*{Why Time Complexity Deteriorates}

The key weakness of IDDFS is \textbf{repeated work}. Consider the root node A: it is expanded in \emph{every} iteration ($L{=}0,1,2,3$) — 4 times. Node B is expanded at $L{=}2$ and $L{=}3$ — twice. Only the deepest nodes (depth $L$) are expanded just once.

However, in a tree with branching factor $b$, the \textbf{number of nodes at the deepest level dominates}:
\[
\text{Total expansions} = (d{+}1) \cdot 1 + d \cdot b + (d{-}1) \cdot b^2 + \cdots + 1 \cdot b^d
\]
\[
= b^d \left(1 + \frac{1}{b} + \frac{1}{b^2} + \cdots\right)
\approx b^d \cdot \frac{b}{b-1}
\]

For $b=10$, the overhead is only $\frac{10}{9} \approx 1.11\times$ — a small price for the enormous space savings ($O(bd)$ vs $O(b^d)$). This is why IDDFS is the \textbf{preferred uninformed search strategy} when the state space is large and step costs are uniform.

\subsubsection*{When IDDFS Is NOT Optimal}

IDDFS finds the \textbf{shallowest} goal (fewest edges), not the lowest-cost goal. With non-uniform step costs, a deeper path with lower total cost may exist. In such cases, UCS or A* is required. For the 2024 exam graph, IDDFS found A$\to$B$\to$E$\to$G (cost 16, 3 hops) while UCS found A$\to$C$\to$H$\to$G (cost 13, 3 hops) — both at the same depth, but UCS found the cheaper one because it considers edge costs, not just depth.

\begin{notesbox}
\textbf{Exam summary — IDDFS:}
\begin{itemize}[noitemsep,topsep=2pt]
  \item \textbf{Completeness:} Inherited from BFS — guaranteed to find a solution if one exists.
  \item \textbf{Optimality:} Inherited from BFS — optimal for \emph{uniform} step costs (finds shallowest solution).
  \item \textbf{Space:} Inherited from DFS — $O(bd)$ linear space, practical for large search spaces.
  \item \textbf{Time:} Worse than both BFS and DFS individually — nodes near the root are re-expanded each iteration. However, the overhead is only a constant factor ($\approx \frac{b}{b-1}$), acceptable given the space savings.
\end{itemize}
\end{notesbox}

\subsection{Section A — Q.A-4(a): DFS vs BFS \ [2 marks]}
%----------------------------------------------------------------------

\begin{solutionbox}
\textbf{Question:} What is an AI Technique? Write down the advantages and disadvantages of Depth-First Search and Breadth-First Search.
\end{solutionbox}

\subsubsection*{AI Technique}

An \textbf{AI Technique} is a method that exploits knowledge, represented in a form that:
\begin{enumerate}[noitemsep]
  \item \textbf{Captures generalisations} — groups situations sharing important properties together, avoiding separate representations for every individual case.
  \item \textbf{Can be understood} by the people providing it — knowledge must be inspectable and modifiable.
  \item \textbf{Can be easily modified} to correct errors and reflect new information.
  \item \textbf{Can be used in many situations} even if not perfectly complete or accurate.
  \item \textbf{Reduces the search space} by narrowing the range of possibilities that must be considered.
\end{enumerate}

\noindent In short: an AI technique is a \textbf{knowledge-driven method for controlling search} so that solutions are found efficiently without exhaustive enumeration of all possibilities. DFS and BFS are two fundamental \emph{uninformed} (blind) search techniques — they use no domain-specific knowledge, only the structure of the search tree.

\subsubsection*{Depth-First Search (DFS)}

\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{p{5.5cm}|p{5.5cm}}
\toprule
\multicolumn{1}{c|}{\textbf{Advantages}} & \multicolumn{1}{c}{\textbf{Disadvantages}} \\
\midrule
\textbf{1. Low memory:} Stores only the current path ($b$ siblings per level), giving $O(bm)$ space. For deep trees, this is the decisive advantage. &
\textbf{1. Not complete:} Fails on infinite-depth trees. DFS may follow an infinite branch forever and never find a solution elsewhere. \\
\midrule
\textbf{2. Simple implementation:} Easy to code recursively or with a stack (LIFO). &
\textbf{2. Not optimal:} The first solution found may be very deep in the tree. DFS has no mechanism to prefer shorter paths. \\
\midrule
\textbf{3. Fast to a solution if lucky:} If the goal is deep and branching is low, DFS may find it with very few node expansions. &
\textbf{3. Susceptible to loops:} Without a closed/visited set, DFS can cycle infinitely. \\
\midrule
\textbf{4. Backtracking-friendly:} Natural fit for constraint satisfaction and puzzle-solving where partial solutions are built and abandoned. &
\textbf{4. May miss shallow goals:} DFS goes deep first — a goal at depth 2 may be found only after exhausting paths to depth 50. \\
\bottomrule
\end{tabular}
\end{center}

\subsubsection*{Breadth-First Search (BFS)}

\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{p{5.5cm}|p{5.5cm}}
\toprule
\multicolumn{1}{c|}{\textbf{Advantages}} & \multicolumn{1}{c}{\textbf{Disadvantages}} \\
\midrule
\textbf{1. Complete:} Guaranteed to find a solution if one exists (provided $b$ is finite). & \textbf{1. Exponential memory:} Must store the entire frontier — $O(b^d)$ space. For $b{=}10$, $d{=}12$: 1 trillion nodes in memory. This makes BFS \textbf{infeasible} for most real problems. \\
\midrule
\textbf{2. Optimal (uniform cost):} Finds the shallowest solution. If step costs are all equal, this is the optimal cost. &
\textbf{2. Exponential time:} $O(b^d)$ — same as the space requirement. Practical only for shallow trees. \\
\midrule
\textbf{3. Systematic:} Processes the entire tree level by level — no path is ever skipped or overlooked. &
\textbf{3. Slow to find deep goals:} Must explore all nodes at depth $d{-}1$ before checking any at depth $d$. If the goal is at depth 10, all 9 shallow levels must be fully expanded first. \\
\midrule
\textbf{4. Good for shallow goals:} If the goal is near the root, BFS finds it quickly. &
\textbf{4. Queue implementation overhead:} Maintaining the frontier as a FIFO queue with $O(1)$ enqueue/dequeue while also tracking visited states adds complexity. \\
\bottomrule
\end{tabular}
\end{center}

\subsubsection*{When to Use Which}

\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|l|l}
\toprule
\textbf{Criterion} & \textbf{Use DFS} & \textbf{Use BFS} \\
\midrule
Memory is limited & \textbf{Yes} — $O(bm)$ space & No — $O(b^d)$ space \\
Goal is known to be deep & \textbf{Yes} — goes deep fast & No — explores all shallow levels first \\
Goal is known to be shallow & No — may waste time deep & \textbf{Yes} — finds it early \\
Need shortest path & No — not optimal & \textbf{Yes} (uniform costs) \\
Infinite / very deep tree & No — may not terminate & \textbf{Yes} — complete \\
\bottomrule
\end{tabular}
\end{center}

\begin{notesbox}
\textbf{The IDDFS resolution:} Iterative Deepening DFS combines the space advantage of DFS ($O(bd)$ memory) with the completeness and optimality of BFS. For most practical uninformed search problems where memory is the bottleneck, \textbf{IDDFS is the preferred choice}. It was tested in 2024 Q2, 2023 Q2(c), 2021 Q3(b), and 2020 Q3(b) — a core exam pattern.
\end{notesbox}

\subsection{Q2(a) — State Space Search Problem \ [0.75 mark]}
%----------------------------------------------------------------------

\begin{solutionbox}
\textbf{Question:} What is the state space search problem?
\end{solutionbox}

A \textbf{state space search problem} is a formal representation of a problem as a directed graph where:
\begin{itemize}[noitemsep]
  \item \textbf{States} are vertices — each represents a possible configuration of the world.
  \item \textbf{Actions} (or operators) are directed edges — transitions from one state to another.
  \item A \textbf{solution} is a sequence of actions (a path) from an initial state to a goal state.
  \item The search algorithm \textbf{systematically explores} the state space to find this path.
\end{itemize}

\noindent Formally, a state space search problem is defined by four components:
\begin{enumerate}[noitemsep]
  \item \textbf{Initial state} — the starting configuration.
  \item \textbf{Goal test} — a function that determines whether a given state is a goal.
  \item \textbf{Successor function} (operators) — given a state, returns all (action, next\_state) pairs.
  \item \textbf{Path cost function} — assigns a numeric cost to each path (typically additive).
\end{enumerate}

\noindent The \textbf{working principle} of a basic search algorithm:
\begin{enumerate}[noitemsep]
  \item Initialise the fringe with the start state.
  \item Loop: remove a state from the fringe; if it satisfies the goal test, return the solution path; otherwise, expand it (generate successors) and add them to the fringe.
  \item Repeat until the fringe is empty (failure) or goal is found (success).
\end{enumerate}

%----------------------------------------------------------------------
\subsection{Q2(b) — Problem Formulations \ [3 marks]}
%----------------------------------------------------------------------

\begin{solutionbox}
\textbf{Question:} Give the initial state, goal test, successor function, cost function, and working principle of a basic search algorithm for each problem.
\end{solutionbox}

\subsubsection*{(i) Map Coloring Problem}

\textbf{Problem:} Using only 4 colors, color a planar map such that no two adjacent regions share the same color.

\begin{description}[leftmargin=3.6cm, labelwidth=3.4cm, labelsep=0.2cm]
  \item[\textbf{State:}]
    A partial assignment of colors $\{R,G,B,Y\}$ to a subset of regions $\{r_1, r_2, \ldots, r_N\}$. Unassigned regions are marked as $\varnothing$.
    \emph{Example:} $\{r_1{=}R,\; r_2{=}G,\; r_3{=}\varnothing, \ldots\}$.

  \item[\textbf{Initial state:}]
    All $N$ regions unassigned: $\{r_i{=}\varnothing \mid i=1,\ldots,N\}$.

  \item[\textbf{Goal test:}]
    All regions are assigned a color, and for every pair of adjacent regions $(r_i, r_j)$, $color(r_i) \neq color(r_j)$.

  \item[\textbf{Successor function:}]
    Pick an unassigned region $r_i$ and assign it any of the 4 colors $\{R,G,B,Y\}$ that is \textbf{not} already used by any of $r_i$'s adjacent (already-colored) regions. If no color is valid, backtrack.

  \item[\textbf{Path cost:}]
    Typically 1 per assignment (uniform cost). Since all solutions assign exactly $N$ colors, any valid solution has cost $N$. The goal is to find \emph{any} valid coloring, not necessarily optimal.

  \item[\textbf{Working principle:}]
    This is a \textbf{constraint satisfaction problem (CSP)}. Backtracking DFS is the natural fit: assign a color to the next variable, check constraint consistency, backtrack on conflict. DFS is preferred over BFS here because all solutions are at the same depth $N$, and DFS's low memory allows backtracking through a deep search tree.
\end{description}

\subsubsection*{(ii) Monkey \& Bananas Problem}

\textbf{Problem:} A 3-ft monkey is in a room with an 8-ft ceiling. Bananas hang from the ceiling. Two 3-ft crates are available (stackable, movable, climbable). The monkey must get the bananas.

\begin{description}[leftmargin=3.6cm, labelwidth=3.4cm, labelsep=0.2cm]
  \item[\textbf{State:}]
    A 4-tuple $(\mathit{monkey\_pos},\; \mathit{crate1\_pos},\; \mathit{crate2\_pos},\; \mathit{has\_bananas})$ where positions are discrete locations in the room and $\mathit{has\_bananas} \in \{\text{true}, \text{false}\}$. If the monkey is on a crate, its effective height increases by the crate height (3 ft per crate; stacking 2 crates gives 6 ft height). The monkey can reach the bananas only if total height $\geq$ 8 ft.

  \item[\textbf{Initial state:}]
    $(\mathit{floor\_position},\; \mathit{initial\_crate1\_pos},\; \mathit{initial\_crate2\_pos},\; \text{false})$ — monkey on floor, crates at their initial locations, no bananas.

  \item[\textbf{Goal test:}]
    $\mathit{has\_bananas} = \text{true}$.

  \item[\textbf{Successor function (operators):}]
    \begin{enumerate}[noitemsep]
      \item \textbf{Walk}(x $\to$ y): Monkey moves from position $x$ to $y$ on the floor (not on a crate).
      \item \textbf{Push}(crate, $x \to y$): Monkey pushes a crate from $x$ to $y$ (must be on floor, adjacent to crate).
      \item \textbf{Climb}(crate): Monkey climbs onto a crate (must be adjacent to it on the floor).
      \item \textbf{ClimbDown}(): Monkey climbs down from crate to floor.
      \item \textbf{Stack}(crate1, crate2): Place crate1 on top of crate2 (monkey must be on floor, crates adjacent).
      \item \textbf{Grasp}(): Monkey grabs bananas (must be within reach — total height $\geq$ 8 ft).
    \end{enumerate}

  \item[\textbf{Path cost:}]
    Number of actions (1 per action). Goal: find the \textbf{shortest} sequence to get the bananas.

  \item[\textbf{Key insight:}]
    The monkey needs height $\geq$ 8 ft. One crate gives 3+3=6 ft (not enough). Two crates stacked give 3+3+3=9 ft (enough). So the solution requires: walk to crate1, push crate1 under bananas, walk to crate2, push crate2 to same spot, stack crate2 on crate1, climb, grasp.
\end{description}

\subsection{Q3(b) — DFID vs IDA* \ [1.75 marks]}
%----------------------------------------------------------------------

\begin{solutionbox}
\textbf{Question:} Differentiate between DFID and IDA* algorithms. Demonstrate how DFID incorporates the advantages of BFS and DFS with an example.
\end{solutionbox}

\subsubsection*{DFID (Depth-First Iterative Deepening)}

DFID runs depth-limited DFS with cutoff $L=0,1,2,\ldots$ until the goal is found. The cutoff is purely \textbf{structural} — it limits the number of edges (depth) from the root.

\vspace{0.3em}
\noindent\textbf{How DFID combines BFS and DFS advantages:}

\begin{center}
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{l|p{5.5cm}|p{5.5cm}|p{4.5cm}}
\toprule
\textbf{Property} & \textbf{BFS} & \textbf{DFS} & \textbf{DFID (inherits)} \\
\midrule
\textbf{Completeness} & \checkmark\ Complete & \texttimes\ May loop forever & \checkmark\ \textbf{Complete} — from BFS. Each iteration has bounded depth; infinite branches cannot trap it. \\
\midrule
\textbf{Optimality} & \checkmark\ Optimal (uniform cost) & \texttimes\ Not optimal & \checkmark\ \textbf{Optimal for uniform costs} — from BFS. Finds shallowest solution first. \\
\midrule
\textbf{Space} & \texttimes\ $O(b^d)$ & \checkmark\ $O(bm)$ & \checkmark\ \textbf{$O(bd)$} — from DFS. Only stores one path. \\
\bottomrule
\end{tabular}
\end{center}

\noindent\textbf{Example — Binary Tree (branching factor $b=2$):}

\begin{center}
\begin{tikzpicture}[
  level 1/.style={sibling distance=3cm, level distance=1.2cm},
  level 2/.style={sibling distance=1.5cm, level distance=1.2cm},
  every node/.style={draw, circle, minimum size=0.55cm, inner sep=0pt, font=\footnotesize\bfseries},
  goalnode/.style={fill=goalgreen},
]
\node {S}
  child { node {A}
    child { node {C} }
    child { node[goalnode] {G} }
  }
  child { node {B}
    child { node {D} }
    child { node {E} }
  };
\end{tikzpicture}
\end{center}

\vspace{0.3em}
\noindent\textbf{DFID execution on this tree:}
\begin{itemize}[noitemsep]
  \item $L{=}0$: Visit S. Not G.
  \item $L{=}1$: Visit S, A, B. Not G.
  \item $L{=}2$: Visit S, A, C, \textbf{G} $\leftarrow$ Found at depth 2.
\end{itemize}

\noindent\textbf{BFS would:} Store all nodes at depth 2 (4 nodes) in the frontier — $O(b^d)$ memory.
\textbf{DFS would:} Store only the path S$\to$A$\to$C (3 nodes) — $O(bd)$ memory.
\textbf{DFID:} Also stores only the path S$\to$A$\to$C (3 nodes) — $O(bd)$ memory, but with the completeness guarantee of BFS.

\subsubsection*{IDA* (Iterative Deepening A*)}

IDA* is the \textbf{informed} counterpart of DFID. Instead of a depth cutoff, it uses an \textbf{$f$-cost cutoff}:
\begin{itemize}[noitemsep]
  \item Each iteration performs DFS, pruning any node where $f(n) = g(n) + h(n) > \textit{cutoff}$.
  \item The initial cutoff is $f(\text{start}) = h(\text{start})$.
  \item After each iteration, the cutoff is increased to the \textbf{smallest $f$-value that exceeded the previous cutoff}.
  \item The algorithm terminates when the goal is found (optimal) or no nodes remain below $\infty$ cutoff.
\end{itemize}

\subsubsection*{Key Differences}

\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{l|p{6.5cm}|p{6.5cm}}
\toprule
\textbf{Criterion} & \textbf{DFID} & \textbf{IDA*} \\
\midrule
\textbf{Cutoff type} & \textbf{Depth} (structural — number of edges from root) & \textbf{$f$-cost} $= g(n)+h(n)$ (informed — uses heuristic knowledge) \\
\midrule
\textbf{Search type} & Uninformed (blind) & Informed (uses heuristic $h$) \\
\midrule
\textbf{Optimality guarantee} & Optimal for \emph{uniform} edge costs only & Optimal for \emph{any} edge costs, provided $h$ is admissible \\
\midrule
\textbf{Overhead per iteration} & Lower — just checks depth $\leq L$ & Higher — must compute $f(n)=g(n)+h(n)$ for every node visited \\
\midrule
\textbf{When to use} & Unknown/unreliable heuristic, uniform costs & Good admissible heuristic available, non-uniform costs \\
\midrule
\textbf{Space complexity} & $O(bd)$ — stores current path & $O(bd)$ — stores current path (same as DFID) \\
\midrule
\textbf{Cutoff progression} & $L = 0, 1, 2, \ldots$ (linear, fixed step) & $f_\textit{cutoff} = \min\{f(n) \mid n \text{ pruned in last iteration}\}$ (adaptive, data-driven) \\
\bottomrule
\end{tabular}
\end{center}

\begin{notesbox}
\textbf{Exam summary:}
\begin{itemize}[noitemsep,topsep=2pt]
  \item \textbf{DFID} = IDDFS = depth-bounded iterative DFS. Uninformed. Uses depth as cutoff.
  \item \textbf{IDA*} = $f$-cost-bounded iterative DFS. Informed. Uses $f(n)=g(n)+h(n)$ as cutoff.
  \item Both algorithms achieve $O(bd)$ space — their key advantage over BFS and A* (which both require $O(b^d)$ space).
  \item DFID is to BFS what IDA* is to A*: the iterative-deepening, space-efficient version.
  \item In practice, IDA* is preferred for problems with good heuristics and large state spaces (e.g., the 15-puzzle).
\end{itemize}
\end{notesbox}

\subsection{Q2(a) — Formulating a Search Problem \ [1.75 marks]}
%----------------------------------------------------------------------

\begin{solutionbox}
\textbf{Question:} How can you formulate a search problem? Give examples of data structures that can be used to represent a state-space both in explicit and implicit ways.
\end{solutionbox}

\subsubsection*{Formulating a Search Problem}

A search problem is formulated by defining \textbf{four components}:

\begin{enumerate}[noitemsep]
  \item \textbf{Initial state} — the state in which the agent starts. \emph{Example:} In a route-finding problem, the starting city.
  \item \textbf{Goal test} — a function $\text{isGoal}(s) \to \{\text{true}, \text{false}\}$ that determines whether a given state $s$ satisfies the goal condition. Can be an explicit set of goal states or an abstract condition.
  \item \textbf{Successor function} $\text{Succ}(s) \to \{(a, s')\}$ — given a state $s$, returns all (action, resulting-state) pairs reachable by legal actions.
  \item \textbf{Path cost function} $g(p)$ — assigns a numeric cost to a path $p$ (typically the sum of step costs along the path).
\end{enumerate}

\noindent Together, these four components define the \textbf{search space} — a directed graph where vertices are states and edges are actions. A \textbf{solution} is a path from the initial state to any state satisfying the goal test. An \textbf{optimal solution} has the lowest path cost among all solutions.

\subsubsection*{Representing State-Spaces}

\noindent\textbf{1. Explicit Representation (State-Space Graph):}

The entire state-space graph is stored in memory before search begins. Each state and its connections are enumerated.

\textbf{Data structures used:}
\begin{itemize}[noitemsep]
  \item \textbf{Adjacency list:} For each state, store a list of (neighbor, cost) pairs. Memory: $O(|V| + |E|)$.
  \item \textbf{Adjacency matrix:} A $|V| \times |V|$ matrix where entry $[i][j]$ holds the cost of the edge from state $i$ to state $j$ (or $\infty$ if no edge). Memory: $O(|V|^2)$.
  \item \textbf{Edge list:} Store all edges as triples (from\_state, to\_state, cost). Memory: $O(|E|)$.
\end{itemize}

\noindent\textbf{Limitation:} Only practical for \textbf{small} state spaces (e.g., the Romania map with 20 cities). For problems like the 8-puzzle ($9!/2 = 181{,}440$ states) or chess ($\sim 10^{47}$ states), explicit storage is infeasible.

\vspace{0.4em}

\noindent\textbf{2. Implicit Representation (Successor Function):}

The state space is \textbf{generated on demand} during search. Only the current state and the successor function are stored; the full graph is never materialised.

\textbf{Data structures used:}
\begin{itemize}[noitemsep]
  \item \textbf{State representation:} A compact data structure encoding the current configuration. \emph{Examples:} a tuple $(x,y)$ for grid positions; a 2D array for the 8-puzzle board; a set of boolean variables for satisfiability problems.
  \item \textbf{Successor function (as code):} A function that, given a state, \textbf{computes} legal next states algorithmically — no lookup table needed. \emph{Example:} For the Water Jug problem, successors of $(x,y)$ are computed by the 6 pouring/filling rules applied to the current values of $x$ and $y$.
  \item \textbf{Fringe (frontier):} A queue (FIFO for BFS), stack (LIFO for DFS), or priority queue (for UCS/A*) holding states discovered but not yet expanded. Memory depends on the algorithm, not the state-space size.
  \item \textbf{Closed/explored set:} A hash table or set storing states already expanded, preventing redundant work. Uses $O(|V_{\text{expanded}}|)$ memory.
\end{itemize}

\noindent\textbf{Advantage:} Scales to \textbf{enormous} state spaces. The 8-puzzle with $10^5$ states is searched without ever storing the full graph. Chess engines use implicit representation — they generate moves on the fly and search to depth 20+ without enumerating all $10^{47}$ positions.

\subsection{Q2(b) — Problem Formulations \ [2 marks]}
%----------------------------------------------------------------------

\begin{solutionbox}
\textbf{Question:} Give the initial state, goal test, successor function, and cost function for each problem.

\subsubsection*{(i) Travelling Salesman Problem (TSP)}

\begin{description}[leftmargin=3.6cm, labelwidth=3.4cm, labelsep=0.2cm]
  \item[\textbf{State:}]
    $(\ell,\; V)$ where $\ell$ is the current city and $V$ is the set of unvisited cities.

  \item[\textbf{Initial state:}]
    $(\text{start\_city},\; \{\text{all other } N{-}1 \text{ cities}\})$.

  \item[\textbf{Goal test:}]
    $\ell = \text{start\_city}$ \textbf{and} $V = \emptyset$.

  \item[\textbf{Successor function:}]
    $\text{Travel}(c_i \to c_j)$ for any $c_j \in V$ (or $c_j = \text{start\_city}$ if $V = \emptyset$). Removes $c_j$ from $V$.

  \item[\textbf{Path cost:}]
    Sum of road distances travelled. Goal: find the \textbf{minimum-cost} Hamiltonian cycle.

  \item[\textbf{Complexity:}] $(N{-}1)!/2$ distinct tours — exponential. This is the canonical NP-hard problem.
\end{description}

\subsubsection*{(ii) Missionaries \& Cannibals}

\begin{description}[leftmargin=3.6cm, labelwidth=3.4cm, labelsep=0.2cm]
  \item[\textbf{State:}]
    $(M_L, C_L, B)$ where $M_L, C_L \in \{0,1,2,3\}$ are missionaries and cannibals on the left bank, and $B \in \{L,R\}$ is the boat position.

  \item[\textbf{Initial state:}]
    $(3, 3, L)$.

  \item[\textbf{Goal test:}]
    $(0, 0, R)$.

  \item[\textbf{Successor function:}]
    Move 1 or 2 people across the river by boat. Valid combinations: $(1M,0C)$, $(2M,0C)$, $(0M,1C)$, $(0M,2C)$, $(1M,1C)$. After each move, verify the \textbf{safety constraint}: on both banks, $M \geq C$ or $M = 0$.

  \item[\textbf{Path cost:}]
    1 per boat crossing. Goal: \textbf{minimum crossings} (known to be 11).
\end{description}

\subsection{Q3(b) — IDDFS Combines BFS + DFS \ [2 marks]}
%----------------------------------------------------------------------

\begin{solutionbox}
\textbf{Question:} Iterative-deepening (ID) is a type of search algorithm said to combine the benefits of breadth-first and depth-first search. Explain, using a diagram or otherwise, how ID achieves these benefits.
\end{solutionbox}

\vspace{0.5em}

\noindent\textbf{The Problem:} BFS guarantees the shortest path but needs $O(b^d)$ memory — infeasible for large search spaces. DFS uses only $O(bm)$ memory but may never terminate on infinite branches and finds suboptimal deep solutions.

\noindent\textbf{The IDDFS Solution:} Run depth-limited DFS repeatedly with cutoff $L = 0, 1, 2, \ldots$ until the goal is found.

\begin{center}
\begin{tikzpicture}[
  level 1/.style={sibling distance=4.5cm, level distance=1.3cm},
  level 2/.style={sibling distance=2.2cm, level distance=1.3cm},
  level 3/.style={sibling distance=1.1cm, level distance=1.2cm},
  every node/.style={draw, circle, minimum size=0.6cm, inner sep=0pt, font=\footnotesize\bfseries},
  goalnode/.style={fill=goalgreen},
]

\begin{scope}[local bounding box=tree]
\node {A}
  child { node {B}
    child { node {D}
      child { node[goalnode] {G} }
      child { node {H} }
    }
    child { node {E} }
  }
  child { node {C}
    child { node {F} }
    child { node {I} }
  };
\end{scope}
\end{tikzpicture}
\end{center}

\vspace{0.4em}
\begin{center}
\renewcommand{\arraystretch}{1.2}
\small
\begin{tabular}{c|l|l}
\toprule
\textbf{$L$} & \textbf{Nodes visited (DFS order)} & \textbf{Space used} \\
\midrule
0 & A & 1 node (just A) \\
1 & A, B, C & 2 nodes (A$\to$B, then A$\to$C) \\
2 & A, B, D, E, C, F, I & 3 nodes (A$\to$B$\to$D at deepest) \\
3 & A, B, D, \textbf{G} $\leftarrow$ \textbf{found!} & 3 nodes \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.4em}

\noindent\textbf{How IDDFS inherits from each:}

\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{p{4cm}|p{6.5cm}|p{6.5cm}}
\toprule
\textbf{Property} & \textbf{Inherited from BFS} & \textbf{Inherited from DFS} \\
\midrule
\textbf{Completeness} &
  \checkmark\ BFS explores level by level, never trapped by infinite branches. IDDFS does the same — depth limits prevent infinite descent. &
  — \\
\midrule
\textbf{Optimality} &
  \checkmark\ BFS finds the shallowest goal. IDDFS does the same — because it exhausts depth $d{-}1$ before checking depth $d$, the first goal found is shallowest. &
  — \\
\midrule
\textbf{Space} &
  — &
  \checkmark\ DFS uses $O(bm)$ memory — only the current path. IDDFS replicates this: at any moment, only $O(bL)$ nodes are stored (the current path plus siblings). For $b{=}10, L{=}10$: $\sim$100 nodes vs BFS's 10 billion. \\
\midrule
\textbf{Simple implementation} &
  — &
  \checkmark\ Like DFS, IDDFS uses a simple stack or recursion. The outer loop adds minimal complexity. \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.4em}

\noindent\textbf{Cost of the compromise — Time Overhead:}
IDDFS re-expands nodes near the root: node A is expanded $d{+}1$ times, depth-1 nodes are expanded $d$ times, etc. However, the deepest level dominates the node count. The overhead factor is $\frac{b}{b-1}$:

\[
\text{Total work} = b^d \!\left(1 + \frac{1}{b} + \frac{1}{b^2} + \cdots\right)
\approx b^d \cdot \frac{b}{b-1}
\]

For $b=10$, the overhead is only $\frac{10}{9} \approx 1.11\times$ — an acceptable price for reducing memory from $O(b^d)$ to $O(bd)$.

\vspace{0.4em}

\begin{answerbox}
\textbf{IDDFS} = BFS completeness $+$ BFS optimality $+$ DFS space efficiency.
The only trade-off is a modest constant-factor increase in time ($\approx$11\% for $b{=}10$),
making it the \textbf{preferred uninformed search strategy} for large state spaces with uniform costs.
\end{answerbox}

\subsection{Q3(c) — BFS Admissibility \ [1.75 marks]}
%----------------------------------------------------------------------

\begin{solutionbox}
\textbf{Question:} When is breadth-first search an admissible search strategy? Briefly explain.
\end{solutionbox}

\subsubsection*{What ``Admissible'' Means}

In search, an algorithm is \textbf{admissible} if it \textbf{always finds the optimal (lowest-cost) solution} when one exists. The term is most commonly applied to heuristics ($h(n) \leq h^*(n)$), but it can also describe complete search algorithms.

\subsubsection*{When BFS is Admissible}

BFS is admissible \textbf{when all step costs are equal (uniform)}. Under this condition:

\begin{enumerate}[noitemsep]
  \item BFS explores the state space \textbf{level by level}: all nodes at depth 0, then depth 1, then depth 2, \ldots
  \item Since every step costs the same amount (say, 1), the \textbf{depth of a node equals its path cost}: $g(n) = \text{depth}(n)$.
  \item Therefore, the \textbf{first goal node encountered} is at the shallowest depth, which corresponds to the \textbf{minimum path cost}.
  \item BFS returns this shallowest goal — which is optimal.
\end{enumerate}

\subsubsection*{When BFS is NOT Admissible}

BFS is \textbf{not admissible when step costs are non-uniform}. Consider:

\begin{center}
\begin{tikzpicture}[
  every node/.style={draw, circle, minimum size=0.7cm, inner sep=0pt, font=\small\bfseries},
  startnode/.style={fill=startblue},
  goalnode/.style={fill=goalgreen},
  lbl/.style={draw=none, fill=white, font=\footnotesize, inner sep=1pt},
  edge/.style={thick, -{Stealth[length=4pt, width=3pt]}}
]
\node[startnode] (S) at (0,0) {S};
\node            (A) at (-2,-1.5) {A};
\node[goalnode]  (G1) at (2,-1.5) {G$_1$};
\node[goalnode]  (G2) at (-2,-3) {G$_2$};

\draw[edge] (S) -- node[lbl, above] {1} (A);
\draw[edge] (S) -- node[lbl, above] {100} (G1);
\draw[edge] (A) -- node[lbl, left] {1} (G2);
\end{tikzpicture}
\end{center}

\noindent BFS would find G$_1$ first (depth 1, cost 100) and return it as the solution. But the optimal path is S $\to$ A $\to$ G$_2$ (depth 2, cost $1+1=2$). BFS returns the \textbf{shallowest} goal, not the \textbf{cheapest}. The solution is suboptimal, so BFS is not admissible when edge costs differ.

\subsubsection*{Formal Statement}

\begin{answerbox}
\textbf{BFS is admissible iff all step costs are equal.}

When costs are uniform, depth $\propto$ cost, and the shallowest goal is the cheapest.
When costs vary, BFS returns the shallowest goal regardless of cost — this may be suboptimal.
For non-uniform costs, use \textbf{UCS} (Uniform-Cost Search) instead — it expands by $g(n)$ rather than depth, guaranteeing optimality for any non-negative edge costs.
\end{answerbox}

\subsection{Q7(b) — A* Heuristic: Underestimate vs Overestimate \ [2 marks]}
%----------------------------------------------------------------------

\begin{solutionbox}
\textbf{Question:} What will happen if $h'$ underestimates $h$ in the A* algorithm? What will happen if $h'$ overestimates $h$?
\end{solutionbox}

\vspace{0.4em}

\noindent\textbf{Notation clarification:} In standard A* notation, $h(n)$ is the heuristic estimate of the cost from $n$ to the goal, and $h^*(n)$ is the \emph{true} optimal cost. The question uses $h'$ for the heuristic and $h$ for the true cost. We adopt the standard convention below.

\subsubsection*{Case 1: Heuristic UNDERESTIMATES ($h(n) \leq h^*(n)$ — Admissible)}

When the heuristic never overestimates the true cost to the goal (it is \textbf{admissible}):

\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{p{7cm}|p{9cm}}
\toprule
\multicolumn{1}{c|}{\textbf{Consequence}} & \multicolumn{1}{c}{\textbf{Explanation}} \\
\midrule
\textbf{A* is optimal} — it will find the lowest-cost solution. &
The proof: A* expands nodes in order of $f(n)=g(n)+h(n)$. Since $h(n) \leq h^*(n)$, we have $f(n) \leq g(n)+h^*(n) = \text{true total cost through } n$. The first time A* expands a goal node G, $f(G) = g(G) + 0 = g(G)$. For any unexpanded node $n$ on a cheaper path to a different goal, $f(n) \leq \text{true cost of that cheaper path} < g(G)$. So A* would have expanded $n$ before G — contradiction. Therefore the first expanded goal must be optimal. \\
\midrule
\textbf{May expand more nodes} than with a perfect heuristic. &
A severe underestimate (e.g., $h(n)=0$ for all $n$, which reduces A* to UCS) leads to more node expansions. The closer $h$ is to $h^*$, the fewer nodes A* expands. \\
\midrule
\textbf{Still complete} — will find a solution if one exists. &
Admissibility does not affect completeness. A* with any heuristic eventually explores the entire (finite) search space if needed. \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.4em}

\noindent\textbf{Example — admissible (underestimate):} $h(\text{Arad}) = 366 \leq h^*(\text{Arad}) = 418$. A* finds the optimal path Arad $\to$ Sibiu $\to$ Rimnicu $\to$ Pitesti $\to$ Bucharest (cost 418).

\subsubsection*{Case 2: Heuristic OVERESTIMATES ($h(n) > h^*(n)$ for some $n$)}

When the heuristic \textbf{overestimates} the true cost for at least one node (it is \textbf{not admissible}):

\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{p{7cm}|p{9cm}}
\toprule
\multicolumn{1}{c|}{\textbf{Consequence}} & \multicolumn{1}{c}{\textbf{Explanation}} \\
\midrule
\textbf{A* may NOT be optimal} — it can return a suboptimal solution. &
If $h(n)$ overestimates the remaining cost through node $n$, then $f(n)$ is inflated. A* may postpone expanding $n$ in favor of nodes whose $f$-values are lower but lie on suboptimal paths. A suboptimal goal may be expanded and returned before the optimal path is discovered. \\
\midrule
\textbf{The optimal path may be ``hidden.''} &
A* considers nodes in order of $f$-value. An overestimating heuristic makes good nodes look worse than they are (artificially high $f$), so they sit deeper in the fringe while A* explores suboptimal-looking nodes with deceptively low $f$. \\
\midrule
\textbf{Still complete} — overestimation does not affect completeness. &
A* will still find \emph{a} solution (the search is still systematic). It just may not be the \emph{best} one. \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.4em}

\noindent\textbf{Example — inadmissible (overestimate):} Suppose $h(\text{Fagaras}) = 200$, but the true cost from Fagaras to Bucharest is $h^*(\text{Fagaras}) = 99$. Then $f(\text{Fagaras}) = 239 + 200 = 439$, while the path through Pitesti has $f(\text{Pitesti}) = 317 + 101 = 418$. The Fagaras path looks worse and is explored later. If Sibiu$\to$Fagaras$\to$Bucharest (cost $239+99=338$) were actually optimal but $h$ made it look worse, A* might return the Pitesti route (cost 418) — suboptimal.

\subsubsection*{Summary}

\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{l|l|l}
\toprule
\textbf{Property} & \textbf{$h$ underestimates (admissible)} & \textbf{$h$ overestimates (inadmissible)} \\
\midrule
Optimality & \textbf{Guaranteed} $\checkmark$ & \textbf{Not guaranteed} $\times$ \\
Node expansions & More than with exact $h$ & Unpredictable — may expand fewer nodes (but miss optimal) \\
Completeness & Guaranteed $\checkmark$ & Guaranteed $\checkmark$ \\
Example & $h(n)=0$ (UCS — slow but optimal) & $h(n) > h^*(n)$ (fast but potentially wrong) \\
\midrule
\textbf{Design rule} & \textbf{Always make $h$ admissible} & \textbf{Never intentionally overestimate} \\
\bottomrule
\end{tabular}
\end{center}

\begin{notesbox}
\textbf{Exam key point:} The \textbf{admissibility condition} $h(n) \leq h^*(n)$ for all $n$ is the \textbf{single most important property} of a heuristic in A*. It is the necessary and sufficient condition for A* optimality (on tree-search; on graph-search, consistency/monotonicity is also required for optimality without reopening nodes). An admissible heuristic guarantees you won't miss the best path; a well-chosen admissible heuristic (close to $h^*$) makes the search efficient.
\end{notesbox}
