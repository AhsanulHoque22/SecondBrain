\section{Core Theory — Reusable Across Years}

\subsection{Defining a Distributed System}
\begin{solbox}
\textbf{Tanenbaum:} ``A distributed system is a collection of independent computers that appear to its users as a single coherent system.''\\[3pt]
\textbf{Working definition (Lecture-01):} a collection of \textbf{entities}, each \textbf{autonomous, programmable, asynchronous and failure-prone}, that communicate through an \textbf{unreliable communication medium}.
\begin{itemize}[leftmargin=1.4cm, itemsep=1pt]
  \item \textbf{Entity} = a process on a device (PC, PDA, server).
  \item \textbf{Communication medium} = wired or wireless network.
\end{itemize}
\textbf{Examples:} Client--Server (NFS), the Web, the Internet, DNS, P2P overlays (BitTorrent), a cloud (Amazon EC2/S3, Azure), a datacenter (Google).
\end{solbox}

\subsection{The 9 Design Goals (Lecture-01)}
\begin{center}\renewcommand{\arraystretch}{1.25}
\begin{tabularx}{\linewidth}{l X}
\toprule
\textbf{Goal} & \textbf{Question it answers} \\
\midrule
Heterogeneity & Can it handle many types of PCs/devices? \\
Robustness & Is it resilient to host crashes, failures, dropped messages? \\
Availability & Are data + services always there for clients? \\
Transparency & Can it hide internal workings from users? \\
Concurrency & Can the server handle many clients at once? \\
Efficiency & Is the service fast; does it use resources fully? \\
Scalability & Can it grow to millions of nodes without degrading? \\
Security & Can it withstand attacks? \\
Openness & Is the system extensible? \\
\bottomrule
\end{tabularx}\end{center}

\subsection{Key ``Hard'' Issues of Distributed Systems}
\begin{itemize}[leftmargin=1.4cm, itemsep=1pt]
  \item \textbf{No global clock} — no single correct notion of time (\textbf{asynchrony}).
  \item \textbf{Unpredictable failures} — a missing response may be a crashed node, a down network path, or a slow link (\textbf{failure-prone, unreliable}).
  \item \textbf{Highly variable bandwidth} (Kbps $\to$ Tbps) and \textbf{variable latency} (ms $\to$ seconds).
  \item \textbf{Large, variable scale} — from 2 to several million hosts.
\end{itemize}

\subsection{Q1(a) — Define distributed and cloud systems; give examples \texorpdfstring{\qmarks{2+1}}{}}
\begin{qbox}Define distributed and cloud systems. Give appropriate examples.\end{qbox}
\begin{solbox}
\textbf{Distributed system:} a collection of independent computers that appears to users as a single coherent system; entities are autonomous, asynchronous and failure-prone, communicating over an unreliable network. \emph{Examples:} the Web, NFS, DNS, BitTorrent.\\[3pt]
\textbf{Cloud system:} a model for \textbf{on-demand, network access to a shared pool of configurable computing resources} (servers, storage, applications) that are rapidly provisioned and released, billed by consumption. \emph{Examples:} Amazon EC2/S3, Microsoft Azure, Google Cloud.\\[3pt]
\textbf{Link:} a cloud is a large-scale distributed system offered as a metered utility.
\end{solbox}

\subsection{Q1(b) — Tightly vs loosely coupled systems \texorpdfstring{\qmarks{3.25}}{}}
\begin{qbox}Differentiate tightly and loosely coupled systems with appropriate figures.\end{qbox}
\begin{solbox}
\begin{center}\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{\linewidth}{l X X}
\toprule
 & \textbf{Tightly coupled} & \textbf{Loosely coupled} \\
\midrule
Memory & \textbf{Shared} common memory (multiprocessor) & \textbf{Private} local memory each node (multicomputer) \\
Communication & Through shared memory & By \textbf{message passing} over network \\
Clock & Often common clock & Independent clocks \\
Coupling & High; fast, low-latency & Low; slower, scalable \\
Example & SMP / parallel machine & Distributed system / cluster \\
\bottomrule
\end{tabularx}\end{center}
\textbf{Figure (describe in exam):}
\begin{itemize}[leftmargin=1.4cm, itemsep=1pt]
  \item Tightly coupled: several CPUs $\to$ one shared memory block via a common bus.
  \item Loosely coupled: each CPU has its \emph{own} memory; CPUs connected by a network/communication link.
\end{itemize}
\end{solbox}

\subsection{Q1(c) — Distributed vs parallel processing \texorpdfstring{\qmarks{2.5}}{}}
\begin{qbox}In what respect are distributed computing systems better than parallel processing systems?\end{qbox}
\begin{solbox}
\begin{itemize}[leftmargin=1.4cm, itemsep=2pt]
  \item \textbf{Scalability} — distributed systems scale to geographically dispersed, virtually unlimited nodes; parallel systems are limited by a single machine's bus/memory.
  \item \textbf{Resource \& information sharing} — remote resources (files, printers, data) shared across sites; parallel systems share only local resources.
  \item \textbf{Reliability / fault tolerance} — no single point of failure; a crashed node need not stop the system. Parallel systems share memory, so a fault is more likely to be fatal.
  \item \textbf{Cost-effectiveness} — built from commodity machines vs. expensive tightly-coupled hardware.
  \item \textbf{Geographic distribution} — services located near users (lower latency); parallel processing is co-located.
\end{itemize}
\end{solbox}

\subsection{Q2(a) — Location vs relocation vs migration transparency \texorpdfstring{\qmarks{3}}{}}
\begin{qbox}How are location, relocation, and migration transparencies different from each other in distributed computing? Explain with examples.\end{qbox}
\begin{solbox}
\begin{center}\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{\linewidth}{l X X}
\toprule
\textbf{Type} & \textbf{Hides} & \textbf{Example} \\
\midrule
\textbf{Location} & \emph{Where} a resource is physically located & A URL/DNS name gives no hint of the server's physical site. \\
\textbf{Relocation} & That a resource \emph{moved} to another place \textbf{while in use} & A mobile user keeps a connection while the service is moved between servers. \\
\textbf{Migration} & That a resource has \emph{moved} to another location (between accesses) & A file moved to another server; clients still use the same name. \\
\bottomrule
\end{tabularx}\end{center}
\textbf{Key distinction:} migration = resource can move (name unchanged); relocation = resource moves \emph{even while being accessed}; location = you cannot tell where it is at all.
\end{solbox}

\subsection{Q2(c) — Design principles for better performance \texorpdfstring{\qmarks{3}}{}}
\begin{qbox}What are the design principles considered useful for better performance in a distributed system?\end{qbox}
\begin{solbox}
\begin{itemize}[leftmargin=1.4cm, itemsep=2pt]
  \item \textbf{Minimise network traffic / communication} — fewer, batched messages; exploit locality.
  \item \textbf{Caching \& replication} — keep data near where it is used; reduce remote access.
  \item \textbf{Exploit parallelism / concurrency} — overlap computation and communication.
  \item \textbf{Reduce remote-access overhead} — move computation to data when cheaper.
  \item \textbf{Load balancing} — distribute work evenly across nodes.
  \item \textbf{Avoid centralised bottlenecks} — no single server/coordinator that all requests pass through.
\end{itemize}
\end{solbox}

\subsection{Q1(a) — Define a distributed system; real-time examples \texorpdfstring{\qmarks{1.75}}{}}
\begin{qbox}Define a distributed system. Mention a few real-time examples of distributed systems.\end{qbox}
\begin{solbox}
\textbf{Definition:} a collection of independent computers that appears to its users as a single coherent system, communicating by message passing over an unreliable network.\\[3pt]
\textbf{Real-time examples:} air-traffic control, ATM / banking networks, telecom switching, multiplayer online games, sensor networks, the World Wide Web, Google search infrastructure.
\end{solbox}

\subsection{Q1(b) — Advantages over standalone; disadvantages \texorpdfstring{\qmarks{2.5}}{}}
\begin{qbox}Mention the advantages of a distributed computing environment over standalone applications. What are the disadvantages of distributed systems?\end{qbox}
\begin{solbox}
\textbf{Advantages:}
\begin{itemize}[leftmargin=1.4cm, itemsep=1pt]
  \item Resource \& information sharing across sites.
  \item Higher reliability / fault tolerance (no single point of failure).
  \item Better price/performance via commodity hardware.
  \item Scalability and incremental growth.
  \item Geographic distribution $\Rightarrow$ lower latency to users.
\end{itemize}
\textbf{Disadvantages:}
\begin{itemize}[leftmargin=1.4cm, itemsep=1pt]
  \item \textbf{Complexity} — harder to design, debug, and reason about (no global clock).
  \item \textbf{Network dependence} — failures/congestion degrade or stop the system.
  \item \textbf{Security} — more attack surface; data crosses public links.
  \item \textbf{Concurrency \& consistency} problems from shared data.
  \item Troubleshooting and management overhead.
\end{itemize}
\end{solbox}

\subsection{Q1(c) — Scalability + challenges \texorpdfstring{\qmarks{2.5}}{}}
\begin{qbox}Define scalability. What are the challenges involved in designing scalable distributed systems?\end{qbox}
\begin{solbox}
\textbf{Scalability:} the ability of a system to handle a growing amount of work (more nodes, users, or geographic spread) \textbf{without significant loss of performance}. Three dimensions: \textbf{size}, \textbf{geographic}, \textbf{administrative}.\\[3pt]
\textbf{Challenges:}
\begin{itemize}[leftmargin=1.4cm, itemsep=1pt]
  \item \textbf{Centralised components} (single server), \textbf{centralised data} (single DB), \textbf{centralised algorithms} (global state) all become bottlenecks $\Rightarrow$ avoid them.
  \item \textbf{Communication latency} grows over wide-area distances.
  \item \textbf{Keeping replicas consistent} as the system grows.
  \item \textbf{Multiple administrative domains} — conflicting policies, security, trust.
\end{itemize}
\textbf{Techniques:} hiding latency (async/caching), distribution (partitioning), replication.
\end{solbox}

\subsection{Q1(d) — Marshalling/Unmarshalling; ACID \texorpdfstring{\qmarks{2}}{}}
\begin{qbox}Define Marshalling and Unmarshalling. What are the ACID properties?\end{qbox}
\begin{solbox}
\textbf{Marshalling:} packing a collection of data items into a form (a flat byte stream / external representation) suitable for transmission across a network — i.e.\ \textbf{serialization} of parameters/objects before sending.\\
\textbf{Unmarshalling:} the inverse — \textbf{disassembling} the received stream back into the original data structures at the destination.\\[3pt]
\textbf{ACID properties (transactions):}
\begin{center}\renewcommand{\arraystretch}{1.2}
\begin{tabularx}{\linewidth}{l X}
\toprule
\textbf{Atomicity} & All-or-nothing: a transaction either completes fully or has no effect. \\
\textbf{Consistency} & Takes the database from one valid state to another (invariants preserved). \\
\textbf{Isolation} & Concurrent transactions do not interfere; result equals some serial order. \\
\textbf{Durability} & Once committed, effects survive crashes (persisted). \\
\bottomrule
\end{tabularx}\end{center}
\end{solbox}
